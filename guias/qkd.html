L5.2Parte 1 · BB84 con NumPy

Guarda el módulo como bb84_numpy.py. Se usa en las partes siguientes. Cada fotón es una posición de los arrays; la base 0 es la rectilínea y la 1 la diagonal. Medir en la base equivocada se modela como un bit aleatorio, y medir en la correcta devuelve el bit preparado: es exactamente la estadística de la regla de Born para los cuatro estados de BB84.

"""Lab L5 · Parte 1 — BB84 con NumPy (modelo de fotón único ideal)."""
import numpy as np

def h2(x):
    x = np.clip(np.asarray(x, dtype=float), 1e-15, 1 - 1e-15)
    return -x * np.log2(x) - (1 - x) * np.log2(1 - x)

def bb84(n, noise=0.0, p_eve=0.0, rng=None):
    """Devuelve un dict con los arrays del protocolo y el QBER del cribado."""
    rng = np.random.default_rng(rng)
    a_bits = rng.integers(0, 2, n)
    a_bases = rng.integers(0, 2, n)          # 0 = rectilínea (+), 1 = diagonal (x)
    # Eve: interceptación-reenvío con probabilidad p_eve
    eve_on = rng.random(n) < p_eve
    e_bases = rng.integers(0, 2, n)
    e_bits = np.where(e_bases == a_bases, a_bits, rng.integers(0, 2, n))
    s_bits = np.where(eve_on, e_bits, a_bits)    # estado reenviado
    s_bases = np.where(eve_on, e_bases, a_bases)
    # Bob
    b_bases = rng.integers(0, 2, n)
    b_bits = np.where(b_bases == s_bases, s_bits, rng.integers(0, 2, n))
    b_bits ^= (rng.random(n) < noise).astype(b_bits.dtype)   # ruido del canal
    sift = a_bases == b_bases
    errors = a_bits[sift] != b_bits[sift]
    return dict(a_bits=a_bits, a_bases=a_bases, b_bits=b_bits, b_bases=b_bases,
                eve_on=eve_on, e_bases=e_bases, e_bits=e_bits, sift=sift, qber=errors.mean(), n_sift=int(sift.sum()))

def estimate(res, frac=0.2, eps=1e-3, rng=None):
    """Estimación de parámetros sobre una muestra aleatoria del cribado."""
    rng = np.random.default_rng(rng)
    idx = np.flatnonzero(res["sift"])
    k = int(round(frac * idx.size))
    sample = rng.choice(idx, size=k, replace=False)
    qhat = np.mean(res["a_bits"][sample] != res["b_bits"][sample])
    q_up = min(0.5, qhat + np.sqrt(np.log(1 / eps) / (2 * k)))   # cota de Hoeffding
    return k, qhat, q_up

if __name__ == "__main__":
    for p in (0.0, 0.5, 1.0):
        r = bb84(100_000, noise=0.02, p_eve=p, rng=2026)

              f"r=1-2h(Q^)={max(0, 1 - 2 * h2(qhat)):.3f}")

L5.4Parte 3 · Tasa de clave frente a QBER y distancia

Primero, la fracción secreta de Shor–Preskill con reconciliación de eficiencia \(f\) y el umbral que resulta. Después, un modelo sencillo de enlace: fuente ideal de fotón único, fibra de 0,2 dB/km, detector con eficiencia 20 %, probabilidad de cuenta oscura \(10^{-6}\) por ventana y error óptico del 1 %. El QBER con cuentas oscuras es \(Q = (e_{\text{opt}}\,\eta + p_d/2)/(\eta + p_d)\): cada cuenta oscura acierta con probabilidad 1/2.

"""Lab L5 · Parte 3 — Tasa de clave frente a QBER y frente a distancia."""
import numpy as np
import matplotlib
matplotlib.use("Agg")
import matplotlib.pyplot as plt
from bb84_numpy import h2

def rate(q, f=1.0):
    """Fracción secreta por bit cribado (Shor–Preskill con reconciliación de eficiencia f)."""
    return np.maximum(0.0, 1 - h2(q) - f * h2(q))

def threshold(f):
    lo, hi = 1e-9, 0.5
    for _ in range(100):
        mid = (lo + hi) / 2
        lo, hi = (mid, hi) if rate(mid, f) > 0 else (lo, mid)
    return lo

print(" Q      r(f=1)   r(f=1,16)")
for q in (0.00, 0.01, 0.02, 0.03, 0.05, 0.08, 0.10, 0.11):
    print(f"{q:.2f}   {rate(q):.4f}   {rate(q, 1.16):.4f}")
for f in (1.0, 1.1, 1.16, 1.2):
    print(f"umbral con f={f:.2f}: Q* = {threshold(f):.4f}")

# Modelo de distancia: fuente de fotón único ideal, fibra 0,2 dB/km, detector eta_d = 0,2,
# probabilidad de cuenta oscura p_d = 1e-6 por ventana, error óptico e_opt = 1 %.
alpha, eta_d, p_d, e_opt, f = 0.2, 0.2, 1e-6, 0.01, 1.16
L = np.arange(0, 301, 1.0)
eta = eta_d * 10 ** (-alpha * L / 10)
p_click = eta + p_d
Q = (e_opt * eta + 0.5 * p_d) / p_click
R = 0.5 * p_click * rate(Q, f)                 # bits por pulso emitido (cribado 1/2)
with np.errstate(divide="ignore"):
    plob = -np.log2(1 - 10 ** (-alpha * L / 10))  # cota PLOB (canal con pérdidas, sin repetidores)
for d in (10, 50, 100, 150, 200):
    i = int(d)
    print(f"L={d:3d} km  eta={eta[i]:.2e}  QBER={Q[i]:.4f}  R={R[i]:.2e} bit/pulso  PLOB={plob[i]:.2e}")
L_max = L[R > 0].max()
print(f"distancia máxima con este modelo: {L_max:.0f} km")

fig, (a1, a2) = plt.subplots(1, 2, figsize=(10, 3.6))
qq = np.linspace(0, 0.13, 400)
a1.plot(qq, rate(qq), label="f = 1 (Shor–Preskill)"); a1.plot(qq, rate(qq, 1.16), label="f = 1,16")
a1.set_xlabel("QBER"); a1.set_ylabel("bits secretos / bit cribado"); a1.legend()
a2.semilogy(L, np.where(R > 0, R, np.nan), label="BB84 fotón único"); a2.semilogy(L, plob, "--", label="cota PLOB")
a2.set_xlabel("distancia (km)"); a2.set_ylabel("bits / pulso"); a2.legend(); fig.tight_layout()
fig.savefig("key_rate.png", dpi=120); print("guardado key_rate.png")

L5.5Parte 4 (opcional) · BB84 con Qiskit 2.x

Cada fotón es un qubit independiente. Alice prepara \(\ket{0}\), \(\ket{1}\), \(\ket{+}\) o \(\ket{-}\) con x y h; medir en la base diagonal es aplicar h y medir en Z. Eve se modela con una medida intermedia: tras medir, el qubit queda colapsado en el estado que ella reenviaría, así que basta con deshacer su h. Como el circuito solo contiene puertas de Clifford y medidas, el método stabilizer de Aer simula miles de qubits en décimas de segundo.

"""Lab L5 · Parte 4 (opcional) — BB84 con Qiskit 2.x y Aer (simulador de estabilizadores)."""
import numpy as np
from qiskit import QuantumCircuit, ClassicalRegister, QuantumRegister
from qiskit_aer import AerSimulator
from qiskit_aer.primitives import SamplerV2

def bb84_circuit(a_bits, a_bases, b_bases, e_bases=None):
    """Un qubit por fotón. e_bases = None -> sin Eve; si no, Eve mide cada qubit en su base."""
    n = len(a_bits)
    q = QuantumRegister(n, "q")
    bob = ClassicalRegister(n, "bob")
    regs = [q, bob]
    if e_bases is not None:
        eve = ClassicalRegister(n, "eve")
        regs.append(eve)
    qc = QuantumCircuit(*regs)
    for i in range(n):                       # Alice prepara |0>,|1>,|+>,|->
        if a_bits[i]:
            qc.x(i)
        if a_bases[i]:
            qc.h(i)
    qc.barrier()
    if e_bases is not None:                  # Eve: medir en su base = interceptar y reenviar
        for i in range(n):
            if e_bases[i]:
                qc.h(i)
            qc.measure(i, eve[i])
            if e_bases[i]:
                qc.h(i)                      # el estado colapsado es el que "reenvía"
        qc.barrier()
    for i in range(n):                       # Bob mide en su base
        if b_bases[i]:
            qc.h(i)
        qc.measure(i, bob[i])
    return qc

def run(n=1000, eve=False, seed=7):
    rng = np.random.default_rng(seed)
    a_bits, a_bases, b_bases = (rng.integers(0, 2, n) for _ in range(3))
    e_bases = rng.integers(0, 2, n) if eve else None
    qc = bb84_circuit(a_bits, a_bases, b_bases, e_bases)
    sampler = SamplerV2(seed=seed, options={"backend_options": {"method": "stabilizer"}})
    result = sampler.run([qc], shots=1).result()[0]
    bits = result.data.bob.get_bitstrings()[0][::-1]   # Qiskit ordena el bit 0 a la derecha
    b_bits = np.array([int(c) for c in bits])
    sift = a_bases == b_bases
    qber = np.mean(a_bits[sift] != b_bits[sift])
    return int(sift.sum()), qber

if __name__ == "__main__":
    import qiskit, qiskit_aer
    print("qiskit", qiskit.__version__, "| qiskit-aer", qiskit_aer.__version__)
    for eve in (False, True):
        n_sift, qber = run(2000, eve)
        print(f"Eve={'sí' if eve else 'no'}: cribado={n_sift}  QBER={qber:.4f}")

L5.6Parte 5 (opcional) · ETSI GS QKD 014 y clave híbrida

Las aplicaciones no hablan con el hardware QKD: piden claves a su key management entity (KME). Construiremos un KME simulado con la biblioteca estándar que respeta las URL, los verbos y los formatos JSON de ETSI GS QKD 014 V1.1.1 (véase la sección 6.8 del Tema 6) y una aplicación que combina la clave QKD con un secreto ML-KEM-768.

Simplificaciones deliberadas

La norma exige HTTPS con TLS 1.2 o superior y autenticación mutua por certificados; el identificador del SAE se obtiene del certificado. Aquí se usa HTTP en 127.0.0.1 y una cabecera X-SAE-ID que no es parte de la norma. Las claves las genera secrets: no hay QKD detrás. Un único proceso hace de los dos KME.

"""Lab L5 · Parte 5 — KME simulado con la forma de la API ETSI GS QKD 014 V1.1.1.
SOLO PARA LABORATORIO: HTTP sin TLS en localhost. La norma exige HTTPS (TLS >= 1.2) con
autenticación mutua por certificados; aquí la identidad del SAE llega en la cabecera
X-SAE-ID, que NO forma parte de la norma. Las claves se generan con secrets (no hay QKD)."""
import base64, json, secrets, sys, threading, uuid
from http.server import BaseHTTPRequestHandler, ThreadingHTTPServer
from urllib.parse import urlparse, parse_qs

KME_ID = {"SAE_A": "KME_MADRID_1", "SAE_B": "KME_MADRID_2"}
KEY_SIZE, MIN_SIZE, MAX_SIZE, MAX_PER_REQ, MAX_COUNT = 256, 64, 1024, 128, 1000
lock = threading.Lock()
pending = {}          # key_ID -> (key bytes, master SAE, slave SAE)
stored = {("SAE_A", "SAE_B"): MAX_COUNT, ("SAE_B", "SAE_A"): MAX_COUNT}  # bits "disponibles" (simulado)

class Handler(BaseHTTPRequestHandler):
    def _send(self, code, obj):
        body = json.dumps(obj).encode()
        self.send_response(code)
        self.send_header("Content-Type", "application/json")
        self.send_header("Content-Length", str(len(body)))
        self.end_headers()
        self.wfile.write(body)

    def _parse(self):
        u = urlparse(self.path)
        parts = u.path.strip("/").split("/")          # api v1 keys {SAE_ID} {method}
        if len(parts) != 5 or parts[:3] != ["api", "v1", "keys"]:
            return None, None, None
        return parts[3], parts[4], parse_qs(u.query)

    def _body(self):
        n = int(self.headers.get("Content-Length") or 0)
        return json.loads(self.rfile.read(n) or b"{}") if n else {}

    def _caller(self):
        sae = self.headers.get("X-SAE-ID")
        return sae if sae in KME_ID else None

    def do_GET(self):
        self._dispatch(self._parse(), {})

    def do_POST(self):
        try:
            body = self._body()
        except json.JSONDecodeError:
            return self._send(400, {"message": "bad request format"})
        self._dispatch(self._parse(), body)

    def _dispatch(self, parsed, body):
        other, method, qs = parsed
        caller = self._caller()
        if caller is None:
            return self._send(401, {"message": "unauthorized"})
        if other not in KME_ID or method not in ("status", "enc_keys", "dec_keys"):
            return self._send(400, {"message": "bad request format"})
        if method == "status":
            return self._send(200, {
                "source_KME_ID": KME_ID[caller], "target_KME_ID": KME_ID[other],
                "master_SAE_ID": caller, "slave_SAE_ID": other, "key_size": KEY_SIZE,
                "stored_key_count": stored[(caller, other)], "max_key_count": MAX_COUNT,
                "max_key_per_request": MAX_PER_REQ, "max_key_size": MAX_SIZE,
                "min_key_size": MIN_SIZE, "max_SAE_ID_count": 0})
        if method == "enc_keys":                       # llamada del SAE maestro
            number = int(body.get("number", qs.get("number", [1])[0]))
            size = int(body.get("size", qs.get("size", [KEY_SIZE])[0]))
            if size % 8 or not MIN_SIZE <= size <= MAX_SIZE:
                return self._send(400, {"message": "size shall be a multiple of 8"})
            if not 1 <= number <= MAX_PER_REQ:
                return self._send(400, {"message": "bad request format"})
            keys = []
            with lock:
                if stored[(caller, other)] < number:
                    return self._send(503, {"message": "keys are being exhausted"})
                stored[(caller, other)] -= number
                for _ in range(number):
                    kid, key = str(uuid.uuid4()), secrets.token_bytes(size // 8)
                    pending[kid] = (key, caller, other)
                    keys.append({"key_ID": kid, "key": base64.b64encode(key).decode()})
            return self._send(200, {"keys": keys})
        # dec_keys: llamada del SAE esclavo; {other} es el SAE maestro
        ids = [k["key_ID"] for k in body.get("key_IDs", [])] or qs.get("key_ID", [])
        out = []
        with lock:
            for kid in ids:
                entry = pending.get(kid)
                if entry is None or entry[1] != other or entry[2] != caller:
                    return self._send(401 if entry else 400, {"message": "key_ID not available",
                                                              "details": [{"key_ID": kid}]})
            for kid in ids:                            # la clave se entrega una sola vez
                key, _, _ = pending.pop(kid)
                out.append({"key_ID": kid, "key": base64.b64encode(key).decode()})
        return self._send(200, {"keys": out})

    def log_message(self, fmt, *args):
        sys.stderr.write("[KME] " + fmt % args + "\n")

La aplicación híbrida: Bob publica su clave de encapsulado ML-KEM-768; Alice pide una clave QKD a su KME (enc_keys), encapsula contra Bob y deriva \(K = \mathsf{HKDF}\text{-}\mathsf{SHA256}(K_{\text{QKD}} \mathbin\Vert K_{\text{KEM}},\ \mathit{info})\), con info ligado al key_ID y al hash de \(\mathit{ek} \mathbin\Vert \mathit{ct}\). Bob recupera la clave QKD por su ID (dec_keys), desencapsula y deriva lo mismo.

"""Lab L5 · Parte 5 — Clave híbrida QKD + ML-KEM-768 con HKDF-SHA256.
Requiere cryptography >= 47 (ML-KEM) y el KME simulado (kme_mock.py) en marcha."""
import base64, hashlib, json, os, sys, urllib.request, urllib.error
from cryptography.hazmat.primitives import hashes, serialization
from cryptography.hazmat.primitives.kdf.hkdf import HKDF
from cryptography.hazmat.primitives.ciphers.aead import AESGCM
from cryptography.hazmat.primitives.asymmetric.mlkem import MLKEM768PrivateKey

KME = sys.argv[1] if len(sys.argv) > 1 else "http://127.0.0.1:8443"

def call(sae, path, body=None):
    data = None if body is None else json.dumps(body).encode()
    req = urllib.request.Request(f"{KME}/api/v1/keys/{path}", data=data,
                                 headers={"X-SAE-ID": sae, "Content-Type": "application/json"})
    try:
        with urllib.request.urlopen(req) as r:
            return r.status, json.load(r)
    except urllib.error.HTTPError as e:
        return e.code, json.load(e)

def combine(k_qkd, k_kem, key_id, transcript):
    """Combinador: HKDF-Extract/Expand sobre K_QKD || K_KEM, ligado al key_ID y al transcript."""
    info = b"curso-pqc L5 hybrid v1|" + key_id.encode() + b"|" + hashlib.sha256(transcript).digest()
    return HKDF(algorithm=hashes.SHA256(), length=32, salt=None, info=info).derive(k_qkd + k_kem)

# 0) Estado del enlace visto por SAE_A
print("status:", call("SAE_A", "SAE_B/status"))

# 1) Bob (SAE_B) genera su par ML-KEM-768 y publica la clave de encapsulado
bob_sk = MLKEM768PrivateKey.generate()
bob_ek = bob_sk.public_key().public_bytes(serialization.Encoding.Raw, serialization.PublicFormat.Raw)

# 2) Alice (SAE_A, maestro) pide una clave QKD de 256 bits para SAE_B
code, cont = call("SAE_A", "SAE_B/enc_keys", {"number": 1, "size": 256})
key_id, k_qkd_a = cont["keys"][0]["key_ID"], base64.b64decode(cont["keys"][0]["key"])
print(f"enc_keys -> {code} key_ID={key_id} ({len(k_qkd_a) * 8} bits)")

# 3) Alice encapsula contra ek de Bob y deriva la clave híbrida
ss_a, ct = bob_sk.public_key().encapsulate()     # (secreto, texto cifrado)
transcript = bob_ek + ct
k_a = combine(k_qkd_a, ss_a, key_id, transcript)
nonce = os.urandom(12)
msg = AESGCM(k_a).encrypt(nonce, b"Informe confidencial: nivel 3", key_id.encode())
print(f"ek={len(bob_ek)} B  ct={len(ct)} B  K_A={k_a.hex()[:32]}...")

# 4) Bob recibe (key_ID, ct, nonce, msg); pide la clave QKD por su ID y desencapsula
code, cont = call("SAE_B", "SAE_A/dec_keys", {"key_IDs": [{"key_ID": key_id}]})
k_qkd_b = base64.b64decode(cont["keys"][0]["key"])
ss_b = bob_sk.decapsulate(ct)
k_b = combine(k_qkd_b, ss_b, key_id, bob_ek + ct)
print(f"dec_keys -> {code}  K_B={k_b.hex()[:32]}...  iguales={k_a == k_b}")
print("descifrado:", AESGCM(k_b).decrypt(nonce, msg, key_id.encode()).decode())

# 5) La clave QKD se entrega una única vez
print("segundo dec_keys ->", call("SAE_B", "SAE_A/dec_keys", {"key_IDs": [{"key_ID": key_id}]}))

L5.9Problemas frecuentes

  • ImportError: cannot import name 'execute' from 'qiskit': código escrito para Qiskit 0.x; usa las primitivas V2 como en la parte 4.
  • ModuleNotFoundError: ... asymmetric.mlkem: versión de cryptography anterior a la 47 o compilada contra un OpenSSL sin ML-KEM. Actualiza o usa liboqs-python como alternativa.
  • OSError: [Errno 98] Address already in use: ya hay un KME escuchando; cambia el puerto o detén el proceso anterior.
  • QBER del 50 % en la parte 1 o 4: se han comparado posiciones no cribadas o el orden de bits está invertido.
